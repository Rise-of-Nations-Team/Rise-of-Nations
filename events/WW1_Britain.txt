							has_war_with = GER
						}
					}
				}

			}
		}
	}
	option = {
		name = ww1_england.9.b		#ignore telegram
		ai_chance = {
			factor = 1
			modifier = {
				OR = {
					has_country_flag = ENG_AI_DEMOCRATIC_HISTORICAL

						rule = ENG_ai_behavior
						option = DEMOCRATIC_HISTORICAL
					}

						is_historical_focus_on = yes
						has_game_rule = {
							rule = ENG_ai_behavior
							option = DEFAULT
						}
