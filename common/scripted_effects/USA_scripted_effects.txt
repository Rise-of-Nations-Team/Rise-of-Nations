# Congress seats stay inside their chamber and drive the congress icons
USA_recalc_congress_icon_frames = {
	clamp_variable = { var = senators_support min = 0 max = senators_total }
	clamp_variable = { var = representatives_support min = 0 max = representatives_total }
	set_temp_variable = { senators_temp = senators_support }
	divide_temp_variable = { senators_temp = senators_total }
	multiply_temp_variable = { senators_temp = 21 }
	if = {
		limit = { check_variable = { senators_temp > 0 } }
		clamp_temp_variable = { var = senators_temp min = 1 max = 21 }
	}
	else = {
		set_temp_variable = { senators_temp = 1 }
	}
	set_variable = { current_senate_icon_frame = senators_temp }
	set_temp_variable = { house_temp = representatives_support }
	divide_temp_variable = { house_temp = representatives_total }
	multiply_temp_variable = { house_temp = 21 }
	if = {
		limit = { check_variable = { house_temp > 0 } }
		clamp_temp_variable = { var = house_temp min = 1 max = 21 }
	}
	else = {
		set_temp_variable = { house_temp = 1 }
	}
	set_variable = { current_house_icon_frame = house_temp }
}

# Swap both chambers over to the incoming party
USA_flip_congress_support = {
	set_variable = { var = USA_congress_flip_buffer value = senators_total }
	subtract_from_variable = { var = USA_congress_flip_buffer value = senators_support }
	set_variable = { var = senators_support value = USA_congress_flip_buffer }
	set_variable = { var = USA_congress_flip_buffer value = representatives_total }
	subtract_from_variable = { var = USA_congress_flip_buffer value = representatives_support }
	set_variable = { var = representatives_support value = USA_congress_flip_buffer }
	clear_variable = USA_congress_flip_buffer
	USA_recalc_congress_icon_frames = yes
}

}

# Open a house vote with the Republicans in favour
setup_gop_vote = {
	set_country_flag = vote_in_progress
	set_variable = { var = votes_required_house value = representatives_total }
	divide_variable = { votes_required_house = 2 }
	set_variable = { var = HOUSE_party2_seatsfor value = HOUSE_party2_seats }
	set_variable = { HOUSE_party3_seatsfor = 0 }
	set_variable = { republican_against_vote = 0 }
	set_variable = { var = democratic_against_vote value = HOUSE_party3_seats }
	set_variable = { var = usa_total_vote value = HOUSE_party2_seatsfor }
	set_variable = { var = usa_total_vote_against value = democratic_against_vote }
}

# State scope: shift one house seat to the Republicans
USA_gop_up = {
	add_to_variable = { USA_state_gop_support = 5 }
	clamp_variable = { var = USA_state_gop_support min = 0 max = 100 }
	owner = {
		add_to_variable = { HOUSE_party2_seats = 1 }
		subtract_from_variable = { HOUSE_party3_seats = 1 }
		clamp_variable = { var = HOUSE_party2_seats min = 0 max = representatives_total }
		clamp_variable = { var = HOUSE_party3_seats min = 0 max = representatives_total }
	}
}
