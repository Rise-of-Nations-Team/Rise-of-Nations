ITA_machine_reckon = {

		limit = { has_country_flag = ITA_g_boards }

		else = { set_variable = { ITA_p_boards_cost = 35 } }
	}

		limit = { has_country_flag = ITA_g_chamber_may_dissolve }
		set_variable = { ITA_c_dissolve_cost = 0 }
		else = { set_variable = { ITA_c_dissolve_cost = 75 } }
	}

		limit = { has_country_flag = ITA_s_army_obeys }
		set_variable = { ITA_p_troops_cost = 10 }
		else = { set_variable = { ITA_p_troops_cost = 25 } }

ITA_piazza_month = {
	set_temp_variable = { ITA_drift = -1 }
	if = {
		limit = { has_country_flag = ITA_s_leagues_underground }
		add_to_temp_variable = { var = ITA_drift value = 1 }
	}
	if = {
		limit = { has_country_flag = ITA_s_chamber_prorogued }
		add_to_temp_variable = { var = ITA_drift value = 1 }
	}
	if = {
		limit = {
			has_idea = ITA_r_the_white_leagues
			has_country_flag = ITA_dnc_path_reformist
		}
		add_to_temp_variable = { var = ITA_drift value = 1 }
	}
	if = {
		limit = {
			has_idea = ITA_r_the_white_leagues
			has_country_flag = ITA_dnc_path_concordat
		}
		subtract_from_temp_variable = { var = ITA_drift value = 1 }
	}
	if = {
		limit = {
			has_war = yes
			surrender_progress > 0.1
		}
		add_to_temp_variable = { var = ITA_drift value = 2 }
	}
	if = {
		limit = { has_country_flag = ITA_g_boards }
		subtract_from_temp_variable = { var = ITA_drift value = 2 }
	}
	if = {
		limit = {
			check_variable = {
				var = ITA_majority
				value = 35
				compare = less_than
			}
		}
		add_to_temp_variable = { var = ITA_drift value = 1 }
	}
	if = {
		limit = { has_idea = ITA_s_the_protocol }
		subtract_from_temp_variable = { var = ITA_drift value = 1 }
	}
	if = {
		limit = { has_idea = ITA_a_the_state_of_siege }
		add_to_temp_variable = { var = ITA_drift value = 0.5 }
	}
	if = {
		limit = { has_idea = ITA_a_the_widened_franchise }
		subtract_from_temp_variable = { var = ITA_drift value = 0.5 }
	}
	if = {
		limit = { has_idea = ITA_c_the_arma_reinforced }
		subtract_from_temp_variable = { var = ITA_drift value = 1 }
	}
	add_to_variable = { var = ITA_piazza value = ITA_drift }
	ITA_piazza_bands = yes
}
