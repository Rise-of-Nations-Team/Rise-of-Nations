	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					tag = ITA
					has_completed_focus = ITADNC_h_the_reserve_is_prepared
					NOT = { has_idea = ITA_h_the_depots_are_full }
				}
				add_timed_idea = {
					idea = ITA_h_the_depots_are_full
					days = 730
				}
			}
			if = {
				limit = {
					original_tag = ITA
					has_country_flag = ITA_cw_the_leaders_in_the_fortress
					NOT = { has_country_flag = ITA_cw_the_prisoners_asked }
				}
				country_event = { id = ita_dnc.2985 days = 30 }
			}
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					tag = ITA
					has_country_flag = ITA_h_trieste_agent_waiting_flag
					FROM.FROM = { state = 993 }

				}
				clr_country_flag = ITA_h_trieste_agent_waiting_flag

					name = "Bartolomeo Cossetto"
					traits = { operative_infiltrator }
					bypass_recruitment = no
					available_to_spy_master = yes
					nationalities = { ITA }
				}
			}
			if = {
				limit = {
					original_tag = ITA
					has_country_flag = ITA_r_civil_war
					divisions_in_state = { state = FROM.FROM size > 0 }
					FROM = {
						original_tag = ITA
						has_country_flag = ITA_r_civil_war
						NOT = { tag = ROOT }
					}
				}
				FROM.FROM = {
					set_state_flag = ITA_cw_state_taken

						}

							set_state_flag = ITA_cw_state_taken_heavy
						}

				}
			}

	on_startup = {
		effect = {
			ITA = {
				if = {
					limit = {
						is_ai = yes
						has_start_date < 1910.1.1
						NOT = { has_country_flag = ITA_cw_ai_drawn }
						OR = {
							AND = {
								is_historical_focus_on = no
								has_game_rule = { rule = ITA_dnc_ai_behavior option = DEFAULT }
							}
							has_game_rule = { rule = ITA_dnc_ai_behavior option = RANDOM }
						}
					}
					set_country_flag = ITA_cw_ai_drawn
					random_list = {
						50 = { set_country_flag = ITA_cw_ai_historical }
						15 = { set_country_flag = ITA_cw_ai_crown }
						10 = { set_country_flag = ITA_cw_ai_constitutional }
						10 = { set_country_flag = ITA_cw_ai_comando }
						5 = { set_country_flag = ITA_cw_ai_oath }
						10 = { set_country_flag = ITA_cw_ai_republic }
					}
				}
				if = {
					limit = {
						has_start_date < 1910.1.1
						NOT = { has_variable = ITA_cw_prefects_line }
					}
					set_variable = { ITA_cw_prefects_line = 50 }
					set_variable = { ITA_cw_crown_line = 60 }
					set_variable = { ITA_cw_piedmont_line = 50 }
				}
			}
		}
	}
	on_civil_war_end = {
		effect = {
			if = {
				limit = {
					original_tag = ITA
					FROM = { original_tag = ITA }
					has_global_flag = ITA_cw_has_started
					NOT = { has_global_flag = ITA_cw_civil_war_end }
				}
				set_global_flag = ITA_cw_civil_war_end
				clr_country_flag = ITA_r_civil_war
				every_state = { clr_state_flag = ITA_cw_secedes }
				if = {
					limit = { has_active_mission = ITA_cw_palermo_asks_its_price }
					remove_mission = ITA_cw_palermo_asks_its_price
				}
				country_event = { id = ita_dnc.2982 days = 1 }
			}
			if = {
				limit = {
					original_tag = ITA
					has_country_flag = ITA_r_civil_war
				}
				clr_country_flag = ITA_r_civil_war
			}
			if = {
				limit = {
					original_tag = ITA
					has_idea = ITA_p_the_country_divided
				}
				remove_ideas = ITA_p_the_country_divided
			}
		}
